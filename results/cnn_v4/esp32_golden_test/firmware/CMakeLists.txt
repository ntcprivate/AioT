cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET "esp32" CACHE STRING "ESP32-WROOM-32")
get_filename_component(AIOT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(ESP_NN_OFF_PROBE 1)
set(PROJECT_VER "v4-golden-1")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(v4_golden_only)
if(NOT IDF_VERSION_MAJOR EQUAL 5 OR NOT IDF_VERSION_MINOR EQUAL 4 OR NOT IDF_VERSION_PATCH EQUAL 4)
    message(FATAL_ERROR "Golden experiment requires ESP-IDF 5.4.4")
endif()
target_compile_definitions(__idf_espressif__esp-tflite-micro PUBLIC TFLITE_SINGLE_ROUNDING=0)
target_compile_definitions(__idf_main PRIVATE ECG_CONV_SINGLE_ROUNDING=1 ECG_HOST_TEST=1)
# ECG_HOST_TEST only removes optional arithmetic trace logging in the copied
# adapters. It does not replace ESP32 kernels or change their arithmetic.
get_target_property(kernel_options __idf_espressif__esp-tflite-micro COMPILE_OPTIONS)
get_target_property(kernel_defines __idf_espressif__esp-tflite-micro COMPILE_DEFINITIONS)
if("-DESP_NN" IN_LIST kernel_options OR "ESP_NN" IN_LIST kernel_defines OR "ESP_NN=1" IN_LIST kernel_defines)
    message(FATAL_ERROR "Expected proven reference arithmetic, but ESP-NN is effective")
endif()
message(STATUS "V4 golden: reference_active=1 esp_nn_effective=0 Conv/FC single; stock other kernels")
